  ```tsx
  interface Props { todos: Todo[]; onOpenOverflow: (id: string) => void; }
  export default function DoneSection({ todos, onOpenOverflow }: Props) {
    const [open, setOpen] = useState(false);
    if (!todos.length) return null;
    return (
      <section className="mt-4">
        <button type="button" onClick={() => setOpen(o => !o)}
                aria-expanded={open}
                className="w-full flex items-center justify-between px-4 py-3 text-textDim">
          <span>{t.done} ({todos.length})</span>
          <span data-testid="done-chevron"
                className={`transition-transform duration-200 ${open ? 'rotate-180' : ''} motion-reduce:transition-none`}>
            {/* chevron svg */}
          </span>
        </button>
        <div data-open={open}
             className="grid transition-[grid-template-rows] duration-200 ease-out motion-reduce:transition-none"
             style={{ gridTemplateRows: open ? '1fr' : '0fr' }}>
          <div className="overflow-hidden">
            <ul className="divide-y divide-hairlineSoft">
              {todos.map(t => (
                <li key={t.id}><TodoRow todo={t} onOpenOverflow={onOpenOverflow} /></li>
              ))}
            </ul>
          </div>
        </div>
      </section>
    );
  }
  ```
